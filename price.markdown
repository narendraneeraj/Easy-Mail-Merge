<div id="emFaq2" class="accordion-collapse collapse" data-bs-parent="#emPricingFAQ">

</div>
</div>

<h3 class="accordion-header">
<button class="accordion-button collapsed" type="button" data-bs-toggle="collapse" data-bs-target="#emFaq4" aria-expanded="false" aria-controls="emFaq4">Can I personalize emails using Excel or CSV?</button>

<div id="emFaq4" class="accordion-collapse collapse" data-bs-parent="#emPricingFAQ">

<div id="emFaq5" class="accordion-collapse collapse" data-bs-parent="#emPricingFAQ">
